-- ============================================================
-- House Management System - Database Schema
-- Run this script in MySQL Workbench (or the mysql CLI) to create
-- the database and its tables before running the application.
-- ============================================================

CREATE DATABASE IF NOT EXISTS apartmentmanager;
USE apartmentmanager;

CREATE TABLE apartment (
    apt_no   VARCHAR(50) PRIMARY KEY,
    apt_type VARCHAR(50),
    rent     INT,
    occupied VARCHAR(50)
);

CREATE TABLE management (
    mgt_id       VARCHAR(50) PRIMARY KEY,
    mgt_username VARCHAR(50),
    mgt_password VARCHAR(50)
);

CREATE TABLE tenant (
    t_id           VARCHAR(50) PRIMARY KEY,
    t_password     VARCHAR(50),
    t_name         VARCHAR(50),
    t_phone        VARCHAR(50),
    t_national_id  VARCHAR(50),
    t_lease_length INT,
    apt_no         VARCHAR(50),
    apt_type       VARCHAR(50),
    t_arrear       VARCHAR(50),
    rent           INT,
    FOREIGN KEY (apt_no) REFERENCES apartment(apt_no)
);

CREATE TABLE rent (
    r_id             INT AUTO_INCREMENT PRIMARY KEY,
    r_date           DATE,
    t_id             VARCHAR(50),
    apt_no           VARCHAR(50),
    r_payment_method VARCHAR(50),
    t_arrear         VARCHAR(50),
    rent             INT,
    FOREIGN KEY (t_id) REFERENCES tenant(t_id)
);

CREATE TABLE complaints (
    c_no     VARCHAR(50) PRIMARY KEY,
    apt_no   VARCHAR(50),
    t_id     VARCHAR(50),
    c_type   VARCHAR(50),
    c_text   VARCHAR(250),
    c_status VARCHAR(50),
    FOREIGN KEY (t_id) REFERENCES tenant(t_id)
);
